# --- Core recommender stack --------------------------------------------------
# numpy/scipy are pinned BELOW what a standalone resolve would pick, because vllm
# 0.26 constrains them. Raising these to 2.5.1 / 1.18.0 (the old macOS pins) makes
# pip backtrack and either downgrade them again or fail.
implicit==0.7.3        # ALS (fold-in via recalculate_item) + ranking metrics (NDCG).
                       # CPU-only here: implicit's GPU extension fails to load with an RMM
                       # symbol error on this box, which is why GPU ALS fit was dropped.
numpy==2.3.5           # Array backbone. Pinned: implicit/scipy/pandas/vllm ABI must agree.
scipy==1.17.1          # Sparse matrices (csr_matrix) for the user x item interactions.
pandas==3.0.3          # Parquet -> DataFrame; maps raw IDs -> contiguous integer indices.
pyarrow==25.0.0        # Streaming Parquet writes/reads (notebooks/data_filtering.ipynb).
scikit-learn==1.8.0    # TF-IDF + cosine similarity for the content-based hybrid CF (recsys/cbhcf.py).
tqdm==4.69.0           # Progress bars used internally by implicit's fit / evaluation.
threadpoolctl==3.6.0   # Controls BLAS thread count; keeps implicit's OpenBLAS well-behaved.
matplotlib==3.11.1     # Warm-up curve plots.
wandb==0.26.1          # Experiment tracking for the warm-up sweeps.


# --- GPU acceleration --------------------------------------------------------
# `torch` is deliberately NOT pinned here: vllm dictates it (see below), and pinning
# it independently is how you get an unsatisfiable resolve. It arrives as a vllm
# dependency at 2.11.0+cu130.
#
# recsys/gpu_retrieval.py, recsys/cf.py, recsys/cbhcf.py and recsys/eval.py all import
# torch through FUNCTION-LOCAL imports, so the package still imports on a machine
# without it -- only the retrieval paths need it.
cupy==13.4.1           # cuSPARSE SpMM for recsys/item_space.py's blocked sparse products
                       # (function-local import; falls back to scipy on CPU if absent).


# --- Intervention A: sentence-embedding content space ------------------------
# Encodes the item catalogue once into data/embeddings/ (~12 GB), keyed by parent_asin.
#
# Version warning: transformers 5.x BREAKS several trust_remote_code embedding models --
# Alibaba-NLP/gte-*-en-v1.5 (RoPE index out of bounds) and jinaai/jina-embeddings-v3
# (XLMRobertaLoRA missing all_tied_weights_keys). Both were dropped from the slate for this
# reason, not on merit. See recsys/intervention_a.py.
sentence-transformers==5.6.1
transformers==5.14.1   # Pinned explicitly: vllm and sentence-transformers both depend on it,
                       # and this is the version both were verified against.
einops==0.8.2          # required by several of the custom-architecture encoders


# --- Intervention B: LLM-simulated interactions ------------------------------
# recsys/coldllm.py imports `vllm` LAZILY, inside VLLMColdLLMSimulator.__init__, so Filtering
# Simulation, the work queue, assembly and SyntheticAugmentedDataset all import and run without
# an engine. Only Refining Simulation needs one.
#
# TWO WARNINGS, both learned the hard way:
#
# 1. `pip install vllm` DICTATES the torch version -- 0.26.0 pins torch 2.11.0, which upgraded
#    this box from 2.6.0+cu124 and pulled the whole CUDA 13 stack with it. Check what a vllm
#    version pins before installing it into an environment that already works.
# 2. That upgrade also installs `torchcodec`, which BREAKS sentence-transformers (and so all of
#    Intervention A's encoding): its loader raises RuntimeError, and sentence_transformers guards
#    the import with `except (ImportError, OSError)`, which does not catch it. Nothing in this
#    project uses torchcodec, and it cannot be excluded from a requirements file -- it must be
#    removed AFTER install:
#        pip uninstall -y torchcodec
#
# recsys/coldllm.py absorbs three API/platform differences so a newer or older vllm still works;
# the pin below is for reproducibility, not because the code needs exactly this release:
#   _structured_outputs()   GuidedDecodingParams (<= ~0.9) vs StructuredOutputsParams (0.26)
#   _engine_role()          LLM(task="generate") (<= ~0.10) vs LLM(runner="generate") (0.26)
#   _enable_wsl_pin_memory()  vLLM disables pinned memory whenever it detects WSL. In 0.26 that is
#                             fatal, not slow: is_uva_available() IS is_pin_memory_available(), so
#                             engine init dies with "RuntimeError: UVA is not available". vLLM's
#                             own escape hatch is VLLM_WSL2_ENABLE_PIN_MEMORY=1, which coldllm.py
#                             sets when the WSL2 kernel clears vLLM's >= 4.19.121 floor (this box
#                             runs 6.18.33) and only when the variable is not already set.
vllm==0.26.0           # pulls torch 2.11.0+cu130 and triton 3.6.0
